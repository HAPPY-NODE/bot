SECURITY WARNING:
If you pushed the repo .env, your bot token / admin ID are public!
-> Reset token NOW: https://discord.com/developers (Application -> Bot -> Reset Token)
-> Then fix the repo:
     rm -f .env && git rm --cached .env && echo ".env" > .gitignore \
     && echo "vps_bot.db" >> .gitignore && echo "venv/" >> .gitignore \
     && echo "bot.log" >> .gitignore && echo "vps_bot.log" >> .gitignore \
     && git add -A && git commit -m "remove .env" && git push

============================================
Option A: Real GCE VM / normal Linux (has systemd)
============================================

apt-get update -y \
&& apt-get install -y docker.io git python3 python3-venv python3-pip \
&& systemctl enable --now docker \
&& git clone https://github.com/HAPPY-NODE/bot.git /root/bot \
&& cd /root/bot \
&& python3 -m venv venv \
&& ./venv/bin/pip install --upgrade pip \
&& ./venv/bin/pip install -r requirements.txt \
&& ./venv/bin/python -c "from vps_backend import get_backend; b=get_backend(); print('Backend:', b.name)" \
&& nano /root/bot/.env \
&& cat > /etc/systemd/system/unixbot.service <<'EOF'
[Unit]
Description=UnixBot Discord Bot
After=network.target docker.service

[Service]
User=root
WorkingDirectory=/root/bot
ExecStart=/root/bot/venv/bin/python /root/bot/bot.py
Restart=always
RestartSec=5
Environment=PYTHONUNBUFFERED=1

[Install]
WantedBy=multi-user.target
EOF
systemctl daemon-reload && systemctl enable --now unixbot && systemctl status unixbot

============================================
Option B: Podman / ttyd container (NO systemd)
============================================

apt-get update -y \
&& apt-get install -y git python3 python3-venv python3-pip proot debootstrap \
&& git clone https://github.com/HAPPY-NODE/bot.git /root/bot \
&& cd /root/bot \
&& python3 -m venv venv \
&& ./venv/bin/pip install --upgrade pip \
&& ./venv/bin/pip install -r requirements.txt \
&& sed -i "s|TOKEN=.*|TOKEN=YOUR_NEW_TOKEN|" .env \
&& ./venv/bin/python -c "from vps_backend import get_backend; b=get_backend(); print('Backend:', b.name)" \
&& nohup ./venv/bin/python bot.py > bot.log 2>&1 &
sleep 5 && tail -20 bot.log

# If you see "Backend: proot" -> containers will run in Proot mode (works with blocked Docker)
# If you see "Backend: docker" -> normal Docker mode

# Force a specific backend:
#   VPS_BACKEND=docker ./venv/bin/python bot.py
#   VPS_BACKEND=proot  ./venv/bin/python bot.py

# Check environment:
#   systemd-detect-virt   (podman = inside a container)
#   unshare --mount true  (fails => Docker containers are blocked, bot will use Proot)

# Update bot later:
#   cd /root/bot && git pull && ./venv/bin/pip install -r requirements.txt \
#   && kill $(cat bot.pid 2>/dev/null) 2>/dev/null; nohup ./venv/bin/python bot.py > bot.log 2>&1 & sleep 3 && tail -5 bot.log
